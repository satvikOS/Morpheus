cmake_minimum_required(VERSION 3.20)
project(morpheus_native VERSION 0.8.0 LANGUAGES CXX)
option(MORPHEUS_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
option(MORPHEUS_BUILD_BENCHMARKS "Build engineering microbenchmarks" ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(morpheus_core src/protocol.cpp src/dsp.cpp src/recording.cpp src/sha256.cpp)
target_include_directories(morpheus_core PUBLIC include)
target_compile_options(morpheus_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
if(MORPHEUS_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(morpheus_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(morpheus_core PUBLIC -fsanitize=address,undefined)
endif()
execute_process(COMMAND git rev-parse --short HEAD WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" OUTPUT_VARIABLE MORPHEUS_GIT_REV OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
if(NOT MORPHEUS_GIT_REV)
  set(MORPHEUS_GIT_REV "unknown")
endif()
target_compile_definitions(morpheus_core PUBLIC MORPHEUS_NATIVE_VERSION="${PROJECT_VERSION}" MORPHEUS_GIT_REV="${MORPHEUS_GIT_REV}")
execute_process(COMMAND git status --porcelain WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" OUTPUT_VARIABLE MORPHEUS_WORKTREE_STATE ERROR_QUIET)
if(MORPHEUS_WORKTREE_STATE)
  target_compile_definitions(morpheus_core PUBLIC MORPHEUS_SOURCE_DIRTY=true)
else()
  target_compile_definitions(morpheus_core PUBLIC MORPHEUS_SOURCE_DIRTY=false)
endif()
add_executable(morpheus_gateway src/gateway.cpp)
target_link_libraries(morpheus_gateway PRIVATE morpheus_core)
add_executable(morpheus_protocol_tool tools/protocol_tool.cpp)
target_link_libraries(morpheus_protocol_tool PRIVATE morpheus_core)
enable_testing()
add_executable(morpheus_core_tests tests/core_tests.cpp)
target_link_libraries(morpheus_core_tests PRIVATE morpheus_core)
add_test(NAME native_core COMMAND morpheus_core_tests)
if(MORPHEUS_BUILD_BENCHMARKS)
  add_executable(morpheus_benchmark bench/benchmark.cpp)
  target_link_libraries(morpheus_benchmark PRIVATE morpheus_core)
endif()
