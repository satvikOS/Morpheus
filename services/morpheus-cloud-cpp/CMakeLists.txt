cmake_minimum_required(VERSION 3.20)
project(morpheus_cloud VERSION 0.8.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(morpheus_cloud src/main.cpp)
target_compile_options(morpheus_cloud PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
execute_process(COMMAND git rev-parse --short HEAD WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" OUTPUT_VARIABLE MORPHEUS_GIT_REV OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
if(NOT MORPHEUS_GIT_REV)
  set(MORPHEUS_GIT_REV "unknown")
endif()
target_compile_definitions(morpheus_cloud PRIVATE MORPHEUS_CLOUD_VERSION="${PROJECT_VERSION}" MORPHEUS_GIT_REV="${MORPHEUS_GIT_REV}")
execute_process(COMMAND git status --porcelain WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" OUTPUT_VARIABLE MORPHEUS_WORKTREE_STATE ERROR_QUIET)
if(MORPHEUS_WORKTREE_STATE)
  target_compile_definitions(morpheus_cloud PRIVATE MORPHEUS_SOURCE_DIRTY_JSON="true")
else()
  target_compile_definitions(morpheus_cloud PRIVATE MORPHEUS_SOURCE_DIRTY_JSON="false")
endif()
